RETURNS DECIMAL(7,2)
NOT DETERMINISTIC
READS SQL DATA
BEGIN
DECLARE antiguedad DECIMAL(7,2)

SELECT DATEDIF(CURDATE(), fecha_contratacion) INTO antiguedad 
FROM Empleados
WHERE empleado_id = p_empleado_id;

RETURN (antiguedad / 365,25);
END//

CREATE FUNCTION jefeDpto (p_departamento_id int)
RETURNS VARCHAR(100)
NO DETERMINISTIC
READS SQL DATA
BEGIN

DECLARE v_nombreJefe VARCHAR(100);

SELECT e.nombre INTO v_nombre_jefe
FROM Departamentos d
JOIN Empleados e ON d.jefe_id = e.empleado_id
WHERE d.departamento_id = p_departamento_id;


RETURN IFNULL(v_nombreJefe, 'Sin jefe asignado');

END//

DELIMITER;

SELECT jefeDpto(5) AS Nombre_Jefe_Dpto;

CREATE FUNCTION ventasEntreFechas(p_incio DATE, p_fin DATE)
RETURNS DECIMAL(10,2)
NO DETERMINISTIC
READS SQL DATA
BEGIN

DECLARE v_valorVentas DECIMAL(10,2);

SELECT SUM(valor) INTO v_valorVentas 
FROM Ventas
WHERE fecha_venta BETWEEN p_incio AND p_fin;

RETURN IFNULL(v_valorVentas, 0);

END//

DELIMITER;

SELECT ventasEntreFechas('2025-12-12', '2026-01-01') As TotalVentasRango

CREATE FUNCTION porcentajeSobreDpto (p_empleado_id INT)
RETURNS DECIMAL(10,2)
NOT DETERMINISTIC
READS SQL DATA

BEGIN

DECLARE v_salario DECIMAL(10,2);
DECLARE v_departamento_id INT;
DECLARE v_total_departamento DECIMAL(10,2);

SELECT salario departamento_id INTO v_salario, v_departamento_id
FROM Empleados
WHERE empleado_id = p_empleado_id;

SELECT SUM(salario) INTO v_total_departamento
FROM Empleados
WHERE departamento_id = v_departamento_id;

RETURN IFNULL(v_salario / v_total_departamento * 100), 0);

END//
DELIMITER;

SELECT porcentajeSobreDpto(2);

CREATE FUNCTION ventasPorCliente(p_cliente_id INT) 
RETURNS VARCHAR(100)
NOT DETERMINISTIC
READS SQL DATA

BEGIN 

DECLARE v_ventas INT;

SELECT COUNT(cliente_id) INTO v_ventas
FROM Ventas
WHERE cliente_id = p_cliente_id;

 RETURN "No tiene ventas";
ELSE
 RETURN "Tiene ventas";

CREATE FUNCTION deptoEmpleado(p_empleado_id INT)
RETURNS VARCHAR(100)
NOT DETERMINISTIC
READS SQL DATA

BEGIN

DECLARE v_departamento VARCHAR(100);

SELECT d.nombre_departamento INTO v_departamento
FROM Departamentos d
JOIN Empleados e ON d.departamento_id = e.departamento_id
WHERE e.empleado_id = p_empleado_id;

RETURN IFNULL(v_departamento, "SIN RESULTADO");

END//

DELIMITER;

SELECT deptoEmpleado(5) AS departamentoDelEmpleado;
