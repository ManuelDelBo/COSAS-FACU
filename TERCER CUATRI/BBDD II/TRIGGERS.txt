CREATE TABLE Auditoria_Clientes (
    AuditoriaID INT AUTO_INCREMENT PRIMARY KEY,
    ClienteID INT,
    Accion VARCHAR(10), -- INSERT o UPDATE
    Fecha DATETIME,
    
    Nombre_Anterior VARCHAR(50),
    Apellido_Anterior VARCHAR(50),
    Email_Anterior VARCHAR(100),
    Telefono_Anterior VARCHAR(15),
    
    Nombre_Nuevo VARCHAR(50),
    Apellido_Nuevo VARCHAR(50),
    Email_Nuevo VARCHAR(100),
    Telefono_Nuevo VARCHAR(15)
);

DELIMITER// 

CREATE TRIGGER trg_auditoria_insert_clientes
AFTER INSERT ON Clientes
FOR EACH ROW
BEGIN
    INSERT INTO Auditoria_Clientes (
        ClienteID, Accion, Fecha,
        Nombre_Nuevo, Apellido_Nuevo, Email_Nuevo, Telefono_Nuevo
    )
    VALUES (
        NEW.ClienteID, 'INSERT', NOW(),
        NEW.Nombre, NEW.Apellido, NEW.Email, NEW.Telefono
    );
END //
DELIMITER;

CREATE TRIGGER trg_update_precio_total
AFTER INSERt on DetallesPedido
FOR EACH ROW
BEGIN
DECLARE v_suma DECIMAL(10,2);

SELECT SUM(d.Cantidad * p.Precio) INTO v_suma
FROM DettallesPedido d
JOIN Productos p ON d.ProductoID = p.ProductoID
WHERE d.PedidoID = NEW.PedidoID;

UPDATE Pedidos
SET PrecioTotal = v_suma
WHERE PedidoID = NEW.PedidoIDM

END//
DELIMITER;

CREATE TRIGGER trg_validar_cantidad_detalle
BEFORE INSERT ON DetallesPedido
FOR EACH ROW
BEGIN
    DECLARE stock_actual INT;

    -- Obtener stock del producto
    SELECT Stock INTO stock_actual
    FROM Productos
    WHERE ProductoID = NEW.ProductoID;

    -- Validar cantidad positiva
    IF NEW.Cantidad <= 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'La cantidad debe ser mayor a 0';
    END IF;

    -- Validar contra stock
    IF NEW.Cantidad > stock_actual THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'No hay suficiente stock disponible';
    END IF;

END //
